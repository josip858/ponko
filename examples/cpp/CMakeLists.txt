cmake_minimum_required(VERSION 3.18)
project(ponko_native_examples LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
# Older build invocations may still supply these cache settings.
if(DEFINED CRABBY_LIBRARY_DIR AND NOT DEFINED PONKO_LIBRARY_DIR)
    set(PONKO_LIBRARY_DIR "${CRABBY_LIBRARY_DIR}" CACHE PATH "Rust library directory")
endif()
if(DEFINED CRABBY_INCLUDE_DIR AND NOT DEFINED PONKO_INCLUDE_DIR)
    set(PONKO_INCLUDE_DIR "${CRABBY_INCLUDE_DIR}" CACHE PATH "Native headers")
endif()
set(PONKO_LIBRARY_DIR "${CMAKE_CURRENT_LIST_DIR}/../../target/release" CACHE PATH "Directory containing the Rust cdylib")
set(PONKO_INCLUDE_DIR "${CMAKE_CURRENT_LIST_DIR}/../../include" CACHE PATH "Directory containing ponko.h and ponko.hpp")
# Use only the shared library: linking Rust staticlibs requires platform-specific dependencies.
if(WIN32)
    find_library(PONKO_LIBRARY NAMES ponko_ffi.dll.lib ponko_ffi PATHS "${PONKO_LIBRARY_DIR}" NO_DEFAULT_PATH REQUIRED)
elseif(APPLE)
    find_file(PONKO_LIBRARY NAMES libponko_ffi.dylib PATHS "${PONKO_LIBRARY_DIR}" NO_DEFAULT_PATH REQUIRED)
else()
    find_file(PONKO_LIBRARY NAMES libponko_ffi.so PATHS "${PONKO_LIBRARY_DIR}" NO_DEFAULT_PATH REQUIRED)
endif()
add_library(ponko_ffi INTERFACE)
target_include_directories(ponko_ffi INTERFACE "${PONKO_INCLUDE_DIR}")
target_link_libraries(ponko_ffi INTERFACE "${PONKO_LIBRARY}")
set(CMAKE_BUILD_RPATH "${PONKO_LIBRARY_DIR}")
function(ponko_native target source)
    add_executable(${target} ${source})
    target_link_libraries(${target} PRIVATE ponko_ffi)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
    endif()
    if(WIN32)
        add_custom_command(TARGET ${target} POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${PONKO_LIBRARY_DIR}/ponko_ffi.dll" "$<TARGET_FILE_DIR:${target}>")
    endif()
    # Games find textures in an assets/ folder next to the executable.
    add_custom_command(TARGET ${target} POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_LIST_DIR}/../../assets" "$<TARGET_FILE_DIR:${target}>/assets")
endfunction()
ponko_native(ponko_c_example main.c)
ponko_native(ponko_cpp_example main.cpp)
ponko_native(ponko_c_tests tests.c)
ponko_native(ponko_cpp_tests tests.cpp)
ponko_native(ponko_cpp_bench bench.cpp)
ponko_native(ponko_cpp_boids boids.cpp)
ponko_native(ponko_cpp_guide guide.cpp)
ponko_native(ponko_cpp_battle battle.cpp)

add_test(NAME cpp_example COMMAND ponko_cpp_example)
add_test(NAME c_abi COMMAND ponko_c_tests)
add_test(NAME cpp_wrapper COMMAND ponko_cpp_tests)
add_test(NAME cpp_guide COMMAND ponko_cpp_guide)
add_test(NAME cpp_boids_headless COMMAND ponko_cpp_boids)
set_tests_properties(cpp_boids_headless PROPERTIES ENVIRONMENT "HEADLESS=1;BOIDS=2000;TICKS=30")
add_test(NAME cpp_battle_headless COMMAND ponko_cpp_battle)
set_tests_properties(cpp_battle_headless PROPERTIES ENVIRONMENT "HEADLESS=1;UNITS=2000;TICKS=600")
ponko_native(ponko_c_compatibility compatibility.c)
ponko_native(ponko_cpp_compatibility compatibility.cpp)
add_test(NAME c_compatibility COMMAND ponko_c_compatibility)
add_test(NAME cpp_compatibility COMMAND ponko_cpp_compatibility)

target_include_directories(ponko_c_abi_baseline PRIVATE "${CMAKE_CURRENT_LIST_DIR}/../../crates/ponko_ffi/tests/fixtures/abi")
